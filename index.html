// 1. 게시글 분류 및 렌더링 로직 보완 async function fetchPosts() { const { data: posts, error } = await supabaseClient .from('posts') .select('*') .order('created_at', { ascending: false }); if (error || !posts) return; const appraisalContainer = document.getElementById('appraisal-posts'); const freeContainer = document.getElementById('free-posts'); // 공지사항 상단 고정 정렬 const sortedPosts = posts.sort((a, b) => { const aIsNotice = a.title.includes('[공지]'); const bIsNotice = b.title.includes('[공지]'); if (aIsNotice && !bIsNotice) return -1; if (!aIsNotice && bIsNotice) return 1; return new Date(b.created_at) - new Date(a.created_at); }); // ★ 확실한 필터링 분기 처리 // 명품 감정 탭: [감정요청]이 포함된 글 + [감정요청]이 포함된 공지글 const appraisalPosts = sortedPosts.filter(p => p.title.includes('[감정요청]')); // 자유게시판 탭: [감정요청]이 포함되지 않은 나머지 모든 글 (자유글, 순수 공지글) const freePosts = sortedPosts.filter(p => !p.title.includes('[감정요청]')); if (appraisalContainer) { appraisalContainer.innerHTML = appraisalPosts.length > 0 ? appraisalPosts.map(p => renderPostCard(p, 'appraisal')).join('') : '

등록된 감정 질문이 없습니다.

'; } if (freeContainer) { freeContainer.innerHTML = freePosts.length > 0 ? freePosts.map(p => renderPostCard(p, 'free')).join('') : '

등록된 게시글이 없습니다.

'; } } // 2. 카드 UI 출력 함수 function renderPostCard(post, currentTab) { const isNotice = post.title.includes('[공지]'); // 제목에서 머리말 태그만 깔끔하게 제거 const cleanTitle = post.title .replace('[공지]', '') .replace('[감정요청]', '') .replace('[자유]', '') .trim(); let bodyText = post.content; let imageUrl = ''; if (post.content.includes('[IMG_URL]:')) { const parts = post.content.split('\n\n[IMG_URL]:'); bodyText = parts[0]; imageUrl = parts[1]; } const safeTitle = escapeHtml(cleanTitle); const safeContent = escapeHtml(bodyText); const dateStr = new Date(post.created_at).toLocaleDateString(); return `
${isNotice ? '👑 관리자' : '익명유저'} ${dateStr}

${isNotice ? '공지' : ''} ${safeTitle}

${imageUrl ? `` : ''}

${safeContent}

`; }
